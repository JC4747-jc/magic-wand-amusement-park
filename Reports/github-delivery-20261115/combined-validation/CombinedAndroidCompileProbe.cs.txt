using System;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using UnityEditor;
using UnityEditor.Build.Player;
using UnityEngine;

// Disposable verification harness; not a product build entry point.
public static class CombinedAndroidCompileProbe
{
    [Serializable] sealed class Output
    {
        public string scope="Android managed script compilation only; no IL2CPP, APK or headset acceptance";
        public string unity=Application.unityVersion;
        public string target="Android";
        public string api="https://docs.unity3d.com/cn/6000.0/ScriptReference/Build.Player.PlayerBuildInterface.CompilePlayerScripts.html";
        public Assembly[] assemblies;
    }
    [Serializable] sealed class Assembly { public string name,sha256;public long bytes; }
    public static void Run()
    {
        const string folder=".codex-tmp/android-managed-compile";
        Directory.CreateDirectory(folder);
        var settings=new ScriptCompilationSettings{group=BuildTargetGroup.Android,target=BuildTarget.Android,options=ScriptCompilationOptions.None};
        var result=PlayerBuildInterface.CompilePlayerScripts(settings,folder);
        if(result.assemblies==null||result.assemblies.Count==0)throw new Exception("Android compilation produced no assemblies");
        var entries=result.assemblies.Select(path=>
        {
            if(!File.Exists(path))path=Path.Combine(folder,Path.GetFileName(path));
            if(!File.Exists(path))throw new Exception("Reported assembly missing: "+path);
            using(var stream=File.OpenRead(path))using(var hash=SHA256.Create())
                return new Assembly{name=Path.GetFileName(path),bytes=stream.Length,sha256=BitConverter.ToString(hash.ComputeHash(stream)).Replace("-","").ToLowerInvariant()};
        }).OrderBy(a=>a.name).ToArray();
        foreach(string name in new[]{"MagicMR.dll","Assembly-CSharp.dll"})
            if(!entries.Any(a=>a.name==name))throw new Exception("Required project assembly missing: "+name);
        if(entries.Any(a=>a.name.Contains("ParkInteractionTests")||a.name.Contains(".Editor")))
            throw new Exception("Editor/test assembly unexpectedly entered player compilation");
        File.WriteAllText(folder+"/result.json",JsonUtility.ToJson(new Output{assemblies=entries},true));
        Debug.Log("COMBINED_ANDROID_MANAGED_COMPILE_PASS: "+entries.Length+" assemblies; MagicMR and Assembly-CSharp included; no APK/device acceptance");
    }
}
